procedure ScDeepSortTrain(
    graph: DGLGraph,
    labels: Tensor,
    dim_in: integer,
    dim_hid: integer,
    num_layers: integer,
    num_genes: integer,
    epochs: integer := 300,
    lr: real := 1e-3,
    weight_decay: real := 0,
    val_ratio: real := 0.2,
    batch_size: integer := 500,
    dropout: real := 0,
    device: string := "cpu"
);
begin
    (* Preprocessing *)
    gene_mask := (graph.ndata["cell_id"] != -1);
    cell_mask := (graph.ndata["cell_id"] == -1);
    num_cells := sum(cell_mask);
    num_genes := sum(gene_mask);
    num_labels := max(labels) + 1;

    (* Split indices *)
    perm := random_permutation(num_cells) + num_genes;
    num_val := floor(num_cells * val_ratio);
    val_idx := perm[1..num_val] → device;
    train_idx := perm[num_val+1..end] → device;

    (* Prepare labels *)
    full_labels := -1 * ones(num_genes + num_cells, dtype=long);
    full_labels[-num_cells..end] := labels;
    graph := graph → device;
    graph.ndata["label"] := full_labels → device;

    (* Initialize model and optimizer *)
    model := GNN(dim_in, num_labels, dim_hid, num_layers, num_genes,
                 activation := ReLU(), dropout := dropout) → device;
    sampler := NeighborSampler(fanouts := [-1] * num_layers, edge_dir := "in");
    optimizer := Adam(model.parameters(), lr := lr, weight_decay := weight_decay);
    loss_fn := CrossEntropyLoss(reduction := "sum");

    (* Training loop *)
    max_val_acc := 0;
    best_state := null;
    for epoch := 1 to epochs do
    begin
        (* Compute loss on training set *)
        loss := cal_loss(model, graph, train_idx, sampler, batch_size, loss_fn, optimizer, device);

        (* Evaluate on train and validation sets *)
        train_acc := evaluate(model, graph, train_idx, sampler, batch_size, device);
        (val_correct, val_unsure, val_acc) := evaluate(model, graph, val_idx, sampler, batch_size, device);

        (* Update best model *)
        if val_acc > max_val_acc then
        begin
            max_val_acc := val_acc;
            best_state := deepcopy(model.state_dict());
            save_model(model, optimizer, species, tissue, device);
        end;

        print("Epoch ", epoch, ": Train Acc = ", train_acc,
              ", Loss = ", loss / length(train_idx),
              ", Val Acc = ", val_acc);
    end;

    (* Restore best model *)
    if best_state ≠ null then
        model.load_state_dict(best_state);
end;

procedure cal_loss(
    model: GNN,
    graph: DGLGraph,
    idx: Tensor,
    sampler: NeighborSampler,
    batch_size: integer,
    loss_fn: LossFunction,
    optimizer: Optimizer,
    device: string
): real;
begin
    model.train();
    total_loss := 0;
    total_size := 0;

    dataloader := DataLoader(graph, idx, sampler, batch_size);
    for _, _, blocks in dataloader do
    begin
        blocks := [b → device for b in blocks];
        x := blocks[0].srcdata["features"];
        y_true := blocks[-1].dstdata["label"];
        y_pred := model(blocks, x);

        loss := loss_fn(y_pred, y_true);
        optimizer.zero_grad();
        loss.backward();
        optimizer.step();

        size := blocks[-1].num_dst_nodes();
        total_loss := total_loss + loss.item() * size;
        total_size := total_size + size;
    end;

    return total_loss / total_size;
end;

procedure evaluate(
    model: GNN,
    graph: DGLGraph,
    idx: Tensor,
    sampler: NeighborSampler,
    batch_size: integer,
    device: string,
    unsure_rate: real := 2.0
): tuple[integer, integer, real];
begin
    model.eval();
    total_correct := 0;
    total_unsure := 0;

    dataloader := DataLoader(graph, idx, sampler, batch_size);
    for _, _, blocks in dataloader do
    begin
        blocks := [b → device for b in blocks];
        x := blocks[0].srcdata["features"];
        y_true := blocks[-1].dstdata["label"];
        y_pred := model(blocks, x);

        for i := 1 to length(y_pred) do
        begin
            max_prob := max(y_pred[i]);
            if max_prob < unsure_rate / num_labels then
                total_unsure := total_unsure + 1
            else if argmax(y_pred[i]) = y_true[i] then
                total_correct := total_correct + 1;
        end;
    end;

    acc := total_correct / length(idx);
    return (total_correct, total_unsure, acc);
end;

procedure save_model(
    model: GNN,
    optimizer: Optimizer,
    species: string,
    tissue: string,
    device: string
);
begin
    state := ["model": model.state_dict(), "optimizer": optimizer.state_dict()];
    filename := species + "-" + tissue + ".pt";
    save_path := "saved_models/single_modality/cell_type_annotation/pretrained/" + species + "/models/";
    mkdir_if_not_exists(save_path);
    torch.save(state, save_path + filename);
end;

function predict_proba(
    model: GNN,
    graph: DGLGraph,
    sampler: NeighborSampler,
    batch_size: integer,
    device: string,
    num_labels: integer
): Tensor;
begin
    model.eval();
    cell_mask := (graph.ndata["cell_id"] == -1);
    idx := where(cell_mask) → device;
    graph := graph → device;

    logits := zeros(graph.number_of_nodes(), num_labels);
    dataloader := DataLoader(graph, idx, sampler, batch_size);
    for _, output_nodes, blocks in dataloader do
    begin
        blocks := [b → device for b in blocks];
        x := blocks[0].srcdata["features"];
        logits[output_nodes] := model(blocks, x).detach().cpu();
    end;

    prob := softmax(logits[cell_mask], dim := -1);
    return prob;
end;

function predict(
    prob: Tensor,
    unsure_rate: real := 2.0,
    return_unsure: boolean := false
): tuple[Tensor, Tensor] or Tensor;
begin
    pred := argmax(prob, dim := 1);
    unsure := max(prob, dim := 1) < unsure_rate / num_labels;

    if return_unsure then
        return (pred, unsure)
    else
        return pred;
end;
