begin
  procedure eval_acc(true_labels, model_output, acc);
  begin
    predicted_indices := argmax(model_output, dim := 1);
    num_classes := model_output.shape[1];
    if true_labels.ndim = 1 then
      true_labels_squeezed := true_labels
    else if true_labels.shape[1] = 1 then
      true_labels_squeezed := squeeze(true_labels, dim := 1)
    else
      begin
        print("Warning: true_labels shape suggests it might already be one-hot or multi-label. Assuming conversion needed for standard accuracy.");
        true_labels_squeezed := true_labels
      end;
    true_labels_long := cast(true_labels_squeezed, dtype := long);
    true_one_hot := one_hot(true_labels_long, num_classes := num_classes);
    accuracy := acc(true_one_hot, predicted_indices);
    return accuracy
  end;

      self.name := name;
      self.graph := {};
      self.label := null

    begin
      assert(idx = 0, "This dataset has only one graph");

      return 1
    end;

    function __repr__();

    end
  end;

  function contrastive_loss(z1, z2, temperature := 0.5);
  begin
    z1 := normalize(z1, dim := -1);
    z2 := normalize(z2, dim := -1);
    logits := mm(z1, transpose(z2)) / temperature;
    labels := arange(z1.size(0)).to(z1.device);
    return cross_entropy(logits, labels)
  end;

  class ZINBLoss;
  begin
    procedure ZINBLoss();
    begin
      super()
    end;

    function forward(x, mean, disp, pi, scale_factor, ridge_lambda := 0.0);
    begin
      eps := 1e-10;
      scale_factor := scale_factor[:, None];
      mean := mean * scale_factor;
      t1 := lgamma(disp + eps) + lgamma(x + 1.0) - lgamma(x + disp + eps);
      t2 := (disp + x) * log(1.0 + (mean / (disp + eps))) + (x * (log(disp + eps) - log(mean + eps)));
      nb_final := t1 + t2;
      nb_case := nb_final - log(1.0 - pi + eps);
      zero_nb := pow(disp / (disp + mean + eps), disp);
      zero_case := -log(pi + ((1.0 - pi) * zero_nb) + eps);
      result := where(le(x, 1e-8), zero_case, nb_case);
      if ridge_lambda > 0 then
        result := result + (ridge_lambda * square(pi));
      result := mean(result);
      return result
    end
  end;

  class MLP;
  begin
    field lins;
    field bns;
    field dropout;

    procedure MLP(in_channels, hidden_channels, out_channels, num_layers, dropout := 0.5);
    begin
      super();
      self.lins := ModuleList();
      self.bns := ModuleList();
      if num_layers = 1 then
        self.lins.append(Linear(in_channels, out_channels))
      else
        begin
          self.lins.append(Linear(in_channels, hidden_channels));
          self.bns.append(BatchNorm1d(hidden_channels));
          for i := 1 to num_layers - 2 do
            begin
              self.lins.append(Linear(hidden_channels, hidden_channels));
              self.bns.append(BatchNorm1d(hidden_channels))
            end;
          self.lins.append(Linear(hidden_channels, out_channels))
        end;
      self.dropout := dropout
    end;

    procedure reset_parameters();
    begin
      for lin in self.lins do lin.reset_parameters();
      for bn in self.bns do bn.reset_parameters()
    end;

    function forward(x, edge_index := null);
    begin
      for i := 0 to len(self.lins) - 2 do
        begin
          x := self.lins[i](x);
          x := relu(x, inplace := true);
          x := self.bns[i](x);
          x := dropout(x, p := self.dropout, training := self.training)
        end;
      x := self.lins[-1](x);
      return x
    end
  end;

  class HetConv;
  begin
    procedure HetConv();
    begin
      super()
    end;

    procedure reset_parameters();
    begin
      (* no-op *)
    end;

    function forward(x, adj_t, adj_t2);
    begin
      x1 := matmul(adj_t.cpu(), x.cpu());
      x2 := matmul(adj_t2.cpu(), x.cpu());
      return cat([x1, x2], dim := 1).to(x.device)
    end
  end;

  class ZINBDecoder;
  begin
    field n_dec_3;
    field input_dim;
    field dec_1;
    field dec_2;
    field dec_3;
    field dec_mean;
    field dec_disp;
    field dec_pi;

    procedure ZINBDecoder(input_dim, n_z, n_dec_1 := 128, n_dec_2 := 256, n_dec_3 := 512);
    begin
      super();
      self.n_dec_3 := n_dec_3;
      self.input_dim := input_dim;
      self.dec_1 := Linear(n_z, n_dec_1);
      self.dec_2 := Linear(n_dec_1, n_dec_3);
      self.dec_3 := Linear(n_dec_2, n_dec_3);
      self.dec_mean := Sequential(Linear(self.n_dec_3, self.input_dim), MeanAct());
      self.dec_disp := Sequential(Linear(self.n_dec_3, self.input_dim), DispAct());
      self.dec_pi := Sequential(Linear(self.n_dec_3, self.input_dim), Sigmoid())
    end;

    function forward(z);
    begin
      dec_h1 := relu(self.dec_1(z));
      dec_h3 := relu(self.dec_2(dec_h1));
      _mean := self.dec_mean(dec_h3);
      _disp := self.dec_disp(dec_h3);
      _pi := self.dec_pi(dec_h3);
      return (_mean, _disp, _pi)
    end
  end;

  class MeanAct;
  begin
    procedure MeanAct();
    begin
      super()
    end;

    function forward(x);
    begin
      return clamp(exp(x), min := 1e-5, max := 1e6)
    end
  end;

  class DispAct;
  begin
    procedure DispAct();
    begin
      super()
    end;

    function forward(x);
    begin
      return clamp(softplus(x), min := 1e-4, max := 1e4)
    end
  end;

  class HeteroNet;
  begin
    field feature_embed;
    field convs;
    field bns;
    field dropout;
    field activation;
    field use_bn;
    field conv_dropout;
    field jump;
    field final_project;
    field num_nodes;
    field adj_t;
    field adj_t2;
    field ZINB;
    field device;

    procedure HeteroNet(in_channels, hidden_channels, out_channels, edge_index, num_nodes, num_layers := 2,
                        dropout := 0.5, num_mlp_layers := 1, use_bn := true, conv_dropout := true, dec_dim := [],
                        device := "cpu");
    begin
      super();
      self.feature_embed := MLP(in_channels, hidden_channels, hidden_channels, num_layers := num_mlp_layers,
                               dropout := dropout);
      self.convs := ModuleList();
      self.convs.append(HetConv());
      self.bns := ModuleList();
      self.bns.append(BatchNorm1d(hidden_channels * 2 * len(self.convs)));
      for l := 1 to num_layers - 1 do
        begin
          self.convs.append(HetConv());
          if l <> num_layers - 2 then
            self.bns.append(BatchNorm1d(hidden_channels * 2 * len(self.convs)))
        end;
      self.dropout := dropout;
      self.activation := relu;
      self.use_bn := use_bn;
      self.conv_dropout := conv_dropout;
      self.jump := JumpingKnowledge("cat");
      last_dim := hidden_channels * (power(2, num_layers + 1) - 1);
      self.final_project := Linear(last_dim, out_channels);
      self.num_nodes := num_nodes;
      self.init_adj(edge_index);
      self.ZINB := ZINBDecoder(in_channels, last_dim, n_dec_1 := dec_dim[0], n_dec_2 := dec_dim[1],
                               n_dec_3 := dec_dim[2]);
      self.device := device
    end;

    procedure reset_parameters();
    begin
      self.feature_embed.reset_parameters();
      self.final_project.reset_parameters();
      for bn in self.bns do bn.reset_parameters()
    end;

    procedure init_adj(edge_index);
    begin
      n := self.num_nodes;
      if edge_index is SparseTensor then
        begin
          dev := edge_index.device;
          adj_t := edge_index;
          adj_t := csr_matrix(adj_t.to_scipy());
          adj_t[adj_t > 0] := 1;
          adj_t[adj_t < 0] := 0;
          adj_t := SparseTensor.from_scipy(adj_t).to(dev)
        end
      else if edge_index is Tensor then
        begin
          row, col := edge_index;
          adj_t := SparseTensor(row := col, col := row, value := null, sparse_sizes := (n, n))
        end;
      adj_t.remove_diag(0);
      adj_t2 := matmul(adj_t, adj_t);
      adj_t2.remove_diag(0);
      adj_t2 := csr_matrix(adj_t2.to_scipy()) - csr_matrix(adj_t.to_scipy());
      adj_t2[adj_t2 > 0] := 1;
      adj_t2[adj_t2 < 0] := 0;
      adj_t := SparseTensor.from_scipy(adj_t);
      adj_t2 := SparseTensor.from_scipy(adj_t2);
      adj_t := gcn_norm(adj_t, null, n, add_self_loops := false);
      adj_t2 := gcn_norm(adj_t2, null, n, add_self_loops := false);
      self.adj_t := adj_t.to(edge_index.device);
      self.adj_t2 := adj_t2.to(edge_index.device)
    end;

    function forward(x, edge_index, decoder := false, save_path := null);
    begin
      adj_t := self.adj_t;
      adj_t2 := self.adj_t2;
      x := self.feature_embed(x);
      x := self.activation(x);
      xs := [x];
      if self.conv_dropout then
        x := dropout(x, p := self.dropout, training := self.training);
      for i := 0 to len(self.convs) - 2 do
        begin
          x := self.convs[i](x, adj_t, adj_t2);
          if self.use_bn then
            x := self.bns[i](x);
          xs.append(x);
          if self.conv_dropout then
            x := dropout(x, p := self.dropout, training := self.training)
        end;
      x := self.convs[-1](x, adj_t, adj_t2);
      if self.conv_dropout then
        x := dropout(x, p := self.dropout, training := self.training);
      xs.append(x);
      x := self.jump(xs);
      if not self.conv_dropout then
        x := dropout(x, p := self.dropout, training := self.training);
      if save_path <> null then
        begin
          save(x, save_path & "_embeddings.pt");
          logger.info("Saved embeddings to " & save_path & "_embeddings.pt")
        end;
      if decoder then
        begin
          _mean, _disp, _pi := self.ZINB(x);
          h := self.final_project(x);
          return (h, _mean, _disp, _pi)
        end;
      x := self.final_project(x);
      return x
    end
  end;

  class scHeteroNet;
  begin
    field device;
    field encoder;
    field min_loss;

    procedure scHeteroNet(d, c, edge_index, num_nodes, hidden_channels, num_layers, dropout, use_bn, device,
                          min_loss);
    begin
      super();
      self.device := device;
      self.encoder := HeteroNet(d, hidden_channels, c, edge_index := edge_index, num_nodes := num_nodes,
                                num_layers := num_layers, dropout := dropout, use_bn := use_bn,
                                dec_dim := [32, 64, 128], device := device);
      self.encoder.to(device);
      self.to(device);
      self.min_loss := min_loss
    end;

    procedure reset_parameters();
    begin
      self.encoder.reset_parameters()
    end;

    function preprocessing_pipeline(log_level := "INFO");
    begin
      transforms := [];
      transforms.append(FilterCellsType());
      transforms.append(AnnDataTransform(sc.pp.filter_genes, min_counts := 3));
      transforms.append(FilterCellsScanpy(min_counts := 1));
      transforms.append(
        HighlyVariableGenesLogarithmizedByTopGenes(n_top_genes := 4000, flavor := "cell_ranger"));
      transforms.append(SaveRaw());
      transforms.append(NormalizeTotal());
      transforms.append(UpdateSizeFactors());
      transforms.append(Log1P());
      transforms.append(HeteronetGraph());
      transforms.append(SetConfig({"label_channel": "cell_type"}));
      return Compose(transforms, log_level := log_level)
    end;

    function forward(dataset, save_path := null);
    begin
      x, edge_index := dataset.x.to(self.device), dataset.edge_index.to(self.device);
      return self.encoder(x, edge_index, save_path := save_path)
    end;

    function propagation(e, edge_index, prop_layers := 1, alpha := 0.5);
    begin
      e := e.unsqueeze(1);
      N := e.shape[0];
      row, col := edge_index;
      d := degree(col, N).float();
      d_norm := 1.0 / d[col];
      value := ones_like(row) * d_norm;
      value := nan_to_num(value, nan := 0.0, posinf := 0.0, neginf := 0.0);
      adj := SparseTensor(row := col, col := row, value := value, sparse_sizes := (N, N));
      for _ := 1 to prop_layers do
        e := e * alpha + matmul(adj, e) * (1 - alpha);
      return e.squeeze(1)
    end;

    function two_hop_propagation(e, edge_index, prop_layers := 1, alpha := 0.5);
    begin
      e := e.unsqueeze(1);
      N := e.shape[0];
      row, col := edge_index;
      d := degree(col, N).float();
      d_norm := 1.0 / d[col];
      value := ones_like(row) * d_norm;
      value := nan_to_num(value, nan := 0.0, posinf := 0.0, neginf := 0.0);
      adj := SparseTensor(row := col, col := row, value := value, sparse_sizes := (N, N));
      adj_2hop := adj @ adj;
      for _ := 1 to prop_layers do
        e := e * alpha + matmul(adj_2hop, e) * (1 - alpha);
      return e.squeeze(1)
    end;

    function detect(dataset, node_idx, device, T, use_prop, use_2hop, oodprop, oodalpha);
    begin
      x, edge_index := dataset.x.to(device), dataset.edge_index.to(device);
      logits := self.encoder(x, edge_index);
      if dataset in ("proteins", "ppi") then
        begin
          logits := stack([logits, zeros_like(logits)], dim := 2);
          neg_energy := T * logsumexp(logits / T, dim := -1).sum(dim := 1)
        end
      else
        neg_energy := T * logsumexp(logits / T, dim := -1);
      if use_prop then
        if use_2hop then
          neg_energy := self.two_hop_propagation(neg_energy, edge_index, oodprop, oodalpha)
        else
          neg_energy := self.propagation(neg_energy, edge_index, oodprop, oodalpha);
      return neg_energy[node_idx]
    end;

    function loss_compute(dataset_ind, dataset_ood, criterion, device, use_zinb);
    begin
      x_in, edge_index_in := dataset_ind.x.to(device), dataset_ind.edge_index.to(device);
      x_out, edge_index_out := dataset_ood.x.to(device), dataset_ood.edge_index.to(device);
      train_in_idx, train_ood_idx := dataset_ind.splits["train"], dataset_ood.node_idx;
      logits_in, _mean, _disp, _pi := (
        i[train_in_idx]
        for i in self.encoder(dataset_ind.x.to(device), dataset_ind.edge_index.to(device), decoder := use_zinb));
      logits_out := self.encoder(x_out, edge_index_out);
      pred_in := log_softmax(logits_in, dim := 1);
      sup_loss := criterion(pred_in, (dataset_ind.y.to(device))[train_in_idx].squeeze(1));
      loss := sup_loss;
      return (loss, _mean, _disp, _pi, train_in_idx, logits_in)
    end;

    procedure fit(dataset_ind, dataset_ood_tr, use_zinb, adata, zinb_weight, cl_weight, mask_ratio, criterion,
                  optimizer);
    begin
      self.train();
      optimizer.zero_grad();
      loss, _mean, _disp, _pi, train_idx, logit_in := self.loss_compute(dataset_ind, dataset_ood_tr, criterion,
                                                                        self.device, use_zinb);
      if use_zinb then
        begin
          zinb_loss := ZINBLoss().to(self.device);
          x_raw := adata.raw.X;
          if issparse(x_raw) then
            x_raw := x_raw.toarray();
          x_raw := tensor(x_raw).to(self.device)[train_idx];
          zinb_loss := zinb_loss(x_raw, _mean, _disp, _pi,
                                 tensor(adata.obs.size_factors, device := self.device)[train_idx]);
          loss := loss + (zinb_weight * zinb_loss)
        end;
      if cl_weight <> 0 then
        begin
          X := dataset_ind.x.to(self.device);
          mask1 := (rand_like(X) > mask_ratio).float();
          X_view1 := X * mask1;
          z1 := self.encoder(X_view1, dataset_ind.edge_index.to(self.device));
          cl_loss := contrastive_loss(logit_in, z1);
          loss := loss + (cl_weight * cl_loss)
        end;
      loss.backward();
      optimizer.step();
      return loss
    end;

    function predict_proba(dataset_ind, save_path := null);
    begin
      self.eval();
      with torch.no_grad() do
        begin
          logits := self(dataset_ind, save_path := save_path);
          probabilities := softmax(logits, dim := 1).cpu()
        end;
      return probabilities
    end;

    function predict(dataset_ind);
    begin
      probabilities := self.predict_proba(dataset_ind);
      predicted_labels := argmax(probabilities, dim := 1);
      return predicted_labels
    end;

    procedure evaluate(dataset_ind, dataset_ood_te, criterion, eval_func, display_step, run, results, epoch, loss,
                       dataset, T, use_prop, use_2hop, oodprop, oodalpha);
    begin
      result, test_ind_score, test_ood_score, representations := self.evaluate_detect(
        dataset_ind, dataset_ood_te, criterion, eval_func, self.device, return_score := true, dataset := dataset,
        T := T, use_prop := use_prop, use_2hop := use_2hop, oodprop := oodprop, oodalpha := oodalpha);
      results[run].append(result);
      if result[-1] < self.min_loss then
        begin
          self.min_loss := result[-1];
          selected_run := run;
          selected_representation := representations.copy();
          selected_full_preds := selected_representation.argmax(axis := -1);
          selected_test_ind_score := test_ind_score.numpy().copy();
          selected_test_ood_score := test_ood_score.numpy().copy()
        end;
      if epoch mod display_step = 0 then
        print("Epoch: ", epoch:02d, ", Loss: ", loss:0.4f, ", AUROC: ", 100 * result[0]:0.2f, "%, AUPR: ",
              100 * result[1]:0.2f, "%, FPR95: ", 100 * result[2]:0.2f, "%, Test Score: ", 100 * result[-2]:0.2f, "%");
      return result
    end;

    function score(x, y, idx, score_func := null, return_pred := false, save_path := null, batch := null);
    begin
      y_pred := self.predict_proba(x, save_path := save_path);
      if save_path <> null and batch <> null then
        savez_compressed(save_path & "_predictions.npz", predictions := y.cpu().numpy(), batch := batch);
      func := partial(eval_acc, acc := resolve_score_func(score_func or self._DEFAULT_METRIC));
      score := func(y[idx], y_pred[idx]);
      return (score, y_pred) if return_pred else score
    end;

    function evaluate_detect(dataset_ind, dataset_ood, criterion, eval_func, device, return_score, dataset, T,
                             use_prop, use_2hop, oodprop, oodalpha, score_func := null);
    begin
      self.eval();
      with torch.no_grad() do
        test_ind_score := self.detect(dataset_ind, dataset_ind.splits["test"], device, T, use_prop, use_2hop,
                                      oodprop, oodalpha).cpu();
      if dataset_ood is list then
        begin
          result := [];
          for d in dataset_ood do
            begin
              with torch.no_grad() do
                test_ood_score := self.detect(d, d.node_idx, device, T, use_prop, use_2hop, oodprop,
                                              oodalpha).cpu();
              auroc, aupr, fpr, _ := get_measures(test_ind_score, test_ood_score);
              result := result & [auroc] & [aupr] & [fpr]
            end
        end
      else
        begin
          with torch.no_grad() do
            test_ood_score := self.detect(dataset_ood, dataset_ood.node_idx, device, T, use_prop, use_2hop,
                                          oodprop, oodalpha).cpu();
          auroc, aupr, fpr, _ := get_measures(test_ind_score, test_ood_score);
          result := [auroc] & [aupr] & [fpr]
        end;
      with torch.no_grad() do
        begin
          out := self(dataset_ind).cpu();
          test_idx := dataset_ind.splits["test"];
          test_score := eval_func(dataset_ind.y[test_idx], out[test_idx],
                                  acc := resolve_score_func(score_func or self._DEFAULT_METRIC));
          valid_idx := dataset_ind.splits["valid"];
          if dataset in ("proteins", "ppi") then
            valid_loss := criterion(out[valid_idx], dataset_ind.y[valid_idx].to(float))
          else
            begin
              valid_out := log_softmax(out[valid_idx], dim := 1);
              valid_loss := criterion(valid_out, dataset_ind.y[valid_idx].squeeze(1))
            end;
          result := result & [test_score] & [valid_loss];
          if return_score then
            return (result, test_ind_score, test_ood_score, out.detach().cpu().numpy())
          else
            return result
        end
    end
  end;

  function get_genename(raw_adata);
  begin
    if "gene_id" in raw_adata.var.keys() then
      gene_name := raw_adata.var["gene_id"].values
    else if "symbol" in raw_adata.var.keys() then
      gene_name := raw_adata.var["symbol"].values
    else
      gene_name := raw_adata.var.index;
    return gene_name
  end;

  procedure set_split(data, train_idx := [], val_idx := [], test_idx := []);
  begin
    adata := data.data;
    raw_adata := adata;
    raw_adata.obs["assay"] := "10x 3' v2";
    raw_adata := raw_adata[raw_adata.obs["assay"] = "10x 3' v2"];
    y := argmax(raw_adata.obsm["cell_type"], axis := 1);
    for obsm in raw_adata.obsm.keys() do
      if obsm in ["cell_type"] then
        adata.obs[obsm & "_raw"] := argmax(raw_adata.obsm[obsm].values, axis := 1)
      else
        print("copy", obsm);
        if raw_adata.obsm[obsm] is DataFrame then
          adata.obsm[obsm] := raw_adata.obsm[obsm].values;
    adata.obs["cell"] := y;
    adata.var["gene_name"] := get_genename(raw_adata);
    ood_class := min(Counter(y).items(), key := lambda x: x[1])[0];
    ood_idx := [i for i, value in enumerate(y) if value = ood_class];
    id_idx := [i for i, value in enumerate(y) if value <> ood_class];
    train_idx := [i for i in train_idx if i not in ood_idx];
    val_idx := [i for i in val_idx if i not in ood_idx];
    test_idx := [i for i in test_idx if i not in ood_idx];
    adata.uns["train_idx"] := train_idx;
    adata.uns["val_idx"] := val_idx;
    adata.uns["test_idx"] := test_idx;
    adata.uns["ood_idx"] := ood_idx;
    adata.uns["id_idx"] := id_idx;
    adata.obs["n_counts"] := adata.X.sum(axis := 1)
  end;

  procedure print_statistics(results, run := null);
  begin
    if run <> null then
      begin
        result := 100 * tensor(results[run]);
        ood_result, test_score, valid_loss := result[:, :-2], result[:, -2], result[:, -1];
        argmin := valid_loss.argmin().item();
        print("Run ", run + 1:02d, ":");
        print("Chosen epoch: ", argmin + 1);
        for k := 0 to result.shape[1] div 3 - 1 do
          begin
            print("OOD Test ", k + 1, " Final AUROC: ", ood_result[argmin, k * 3]:0.2f);

            print("OOD Test ", k + 1, " Final FPR95: ", ood_result[argmin, k * 3 + 2]:0.2f)
          end;
        print("IND Test Score: ", test_score[argmin]:0.2f)
      end
    else
      begin
        result := 100 * tensor(results);
        ood_te_num := result.shape[2] div 3;
        best_results := [];
        for r in result do
          begin
            ood_result, test_score, valid_loss := r[:, :-2], r[:, -2], r[:, -1];
            score_val := test_score[valid_loss.argmin()].item();
            ood_result_val := [];
            for k := 0 to ood_te_num - 1 do
              begin
                auroc_val := ood_result[valid_loss.argmin(), k * 3].item();
                aupr_val := ood_result[valid_loss.argmin(), k * 3 + 1].item();
                fpr_val := ood_result[valid_loss.argmin(), k * 3 + 2].item();
                ood_result_val := ood_result_val & [auroc_val, aupr_val, fpr_val]
              end;
            best_results.append(ood_result_val & [score_val])
          end;
        best_result := tensor(best_results);
        if best_result.shape[0] = 1 then
          begin
            print("All runs:");
            for k := 0 to ood_te_num - 1 do
              begin
                r := best_result[:, k * 3];
                print("OOD Test ", k + 1, " Final AUROC: ", r.mean():0.2f);
                r := best_result[:, k * 3 + 1];
                print("OOD Test ", k + 1, " Final AUPR: ", r.mean():0.2f);
                r := best_result[:, k * 3 + 2];
                print("OOD Test ", k + 1, " Final FPR: ", r.mean():0.2f)
              end;
            r := best_result[:, -1];
            print("IND Test Score: ", r.mean():0.2f)
          end
        else
          begin
            print("All runs:");
            for k := 0 to ood_te_num - 1 do
              begin
                r := best_result[:, k * 3];
                print("OOD Test ", k + 1, " Final AUROC: ", r.mean():0.2f, " ± ", r.std():0.2f);
                r := best_result[:, k * 3 + 1];
                print("OOD Test ", k + 1, " Final AUPR: ", r.mean():0.2f, " ± ", r.std():0.2f);
                r := best_result[:, k * 3 + 2];
                print("OOD Test ", k + 1, " Final FPR: ", r.mean():0.2f, " ± ", r.std():0.2f)
              end;
            r := best_result[:, -1];
            print("IND Test Score: ", r.mean():0.2f, " ± ", r.std():0.2f)
          end;
        return best_result
      end
  end;

  function stable_cumsum(arr, rtol := 1e-05, atol := 1e-08);
  begin
    out := cumsum(arr, dtype := float64);
    expected := sum(arr, dtype := float64);
    if not allclose(out[-1], expected, rtol := rtol, atol := atol) then
      raise RuntimeError("cumsum was found to be unstable: its last element does not correspond to sum");
    return out
  end;

  function fpr_and_fdr_at_recall(y_true, y_score, recall_level := 0.95, pos_label := null);
  begin
    classes := unique(y_true);
    if (pos_label = null and not (array_equal(classes, [0, 1]) or array_equal(classes, [-1, 1]) or
                                  array_equal(classes, [0]) or array_equal(classes, [-1]) or
                                  array_equal(classes, [1]))) then
      raise ValueError("Data is not binary and pos_label is not specified")
    else if pos_label = null then
      pos_label := 1.0;
    y_true := (y_true = pos_label);
    desc_score_indices := argsort(y_score, kind := "mergesort")[::-1];
    y_score := y_score[desc_score_indices];
    y_true := y_true[desc_score_indices];
    distinct_value_indices := where(diff(y_score))[0];
    threshold_idxs := r_[distinct_value_indices, y_true.size - 1];
    tps := stable_cumsum(y_true)[threshold_idxs];
    fps := 1 + threshold_idxs - tps;
    thresholds := y_score[threshold_idxs];
    recall := tps / tps[-1];
    last_ind := tps.searchsorted(tps[-1]);
    sl := slice(last_ind, null, -1);
    recall, fps, tps, thresholds := r_[recall[sl], 1], r_[fps[sl], 0], r_[tps[sl], 0], thresholds[sl];
    cutoff := argmin(abs(recall - recall_level));
    if array_equal(classes, [1]) then
      return thresholds[cutoff];
    return (fps[cutoff] / sum(not_(y_true))), thresholds[cutoff]
  end;

  function get_measures(_pos, _neg, recall_level := 0.95);
  begin
    pos := array(_pos[:]).reshape((-1, 1));
    neg := array(_neg[:]).reshape((-1, 1));
    examples := squeeze(vstack((pos, neg)));
    labels := zeros(len(examples), dtype := int32);
    labels[:len(pos)] += 1;
    auroc := roc_auc_score(labels, examples);
    aupr := average_precision_score(labels, examples);
    fpr, threshold := fpr_and_fdr_at_recall(labels, examples, recall_level);
    return (auroc, aupr, fpr, threshold)
  end;
